cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" CTIMER_VERSION LIMIT_COUNT 1)
project(Ctimer VERSION ${CTIMER_VERSION} LANGUAGES CXX RC)
configure_file(resources/version.h.in generated/version.h @ONLY)
configure_file(resources/app.manifest.in generated/app.manifest @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(ctimer_core src/core.cpp src/platform.cpp src/ntp.cpp src/engine.cpp)
target_include_directories(ctimer_core PUBLIC src)
target_compile_definitions(ctimer_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(ctimer_core PUBLIC ws2_32 user32 shell32 advapi32)
add_executable(Ctimer WIN32 src/main.cpp resources/app.rc)
target_include_directories(Ctimer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" resources)
find_program(CTIMER_POWERSHELL NAMES pwsh powershell)
if(CTIMER_POWERSHELL)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_SOURCE_DIR}/resources/app.ico
    COMMAND ${CTIMER_POWERSHELL} -NoProfile -File ${CMAKE_CURRENT_SOURCE_DIR}/scripts/make-icon.ps1
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/icon.png ${CMAKE_CURRENT_SOURCE_DIR}/scripts/make-icon.ps1
    VERBATIM)
  add_custom_target(ctimer_icon DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/resources/app.ico)
  add_dependencies(Ctimer ctimer_icon)
endif()
target_link_libraries(Ctimer PRIVATE ctimer_core comctl32 gdi32 wtsapi32 winmm)
add_executable(CtimerTests tests/tests.cpp)
target_link_libraries(CtimerTests PRIVATE ctimer_core)
if(MSVC)
  target_compile_options(ctimer_core PUBLIC /utf-8 /W4)
else()
  target_compile_options(ctimer_core PUBLIC -Wall -Wextra)
  target_link_options(Ctimer PRIVATE -municode -static)
  target_link_options(CtimerTests PRIVATE -static)
endif()
enable_testing()
add_test(NAME CtimerTests COMMAND CtimerTests --app $<TARGET_FILE:Ctimer>)
